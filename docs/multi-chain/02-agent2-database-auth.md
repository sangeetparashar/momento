### Step 5: Add helper function for resolving wallet address by chain

Add this utility function to `apps/api/lib/auth.ts`:

```typescript
/**
 * Resolves the wallet address for a given chain from the AuthUser.
 * Returns null if the user doesn't have a wallet on that chain.
 */
export function getWalletForChain(user: AuthUser, chain: string): string | null {
  if (chain === 'base') {
    return user.baseWalletAddress || null;
  }
  // Default to Solana
  const addr = user.walletAddress;
  return addr && !addr.startsWith('pending-') ? addr : null;
}

/**
 * Parses the chain parameter from a request.
 * Defaults to 'solana' if not specified.
 */
export function getChainParam(request: NextRequest): 'solana' | 'base' {
  const url = new URL(request.url);
  const chain = url.searchParams.get('chain');
  if (chain === 'base') return 'base';
  return 'solana';
}
```
